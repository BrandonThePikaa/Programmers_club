PICO-8 Cheat Sheet

Game Structure


function _init()   -- runs once at start
end

function _update() -- runs every frame (30fps)
end

function _draw()   -- runs every frame, after _update
end

_init(): setup variables
_update(): game logic, input
_draw(): visuals and UI

Variables & Types


x = 10
name = "bob"
alive = true
player = {x=10, y=20, hp=10}

Lua is dynamic typed. Use tables for objects and lists.

Operators


+ - * / %       -- arithmetic
== != < > <= >=  -- comparison
and or not      -- logic
= += -= *= /=    -- assignment shorthand

Control Flow


-- if statement
if x>10 then
  y += 1
elseif x>5 then
  y += 0
else
  y -= 1
end

-- for loop
for i=1,10 do
  print(i)
end

-- while loop
while x<10 do
  x += 1
end

Functions


function add(a,b)
  return a+b
end

z = add(3,4)

Tables (arrays/objects)


player = {x=10, y=20, hp=10} -- object
enemies = {}
add(enemies, {x=10, y=5})
for e in all(enemies) do
  e.y += 1
end

Use all(t) to iterate, add(t,val) to append, del(t,val) to remove.

Input


btn(0) -- left
btn(1) -- right
btn(2) -- up
btn(3) -- down
btn(4) -- X
btn(5) -- O

btnp(4) -- pressed this frame
btnr(4) -- released this frame

Drawing


cls()                   -- clear screen
rect(x1,y1,x2,y2,color) -- outline rectangle
rectfill(x1,y1,x2,y2,color) -- filled rectangle
circ(x,y,r,color)

line(x1,y1,x2,y2,color)
print("text", x, y, color)

Colors: 0-15

Sprites


spr(n, x, y, w=1, h=1, flip_x=false, flip_y=false)
-- n = sprite index
-- w,h = tiles wide/tall
spr(1, player.x, player.y)

Collision (rectangle)


function collide(a,b)
  return not (
    a.x+a.w < b.x or
    a.x > b.x+b.w or
    a.y+a.h < b.y or
    a.y > b.y+b.h
  )
end

Game States


state="menu"
function _update()
  if state=="menu" and btnp(4) then state="play"
  elseif state=="play" then
    -- game logic
  end
end

Quick Mini-Patterns


-- Move player
if btn(0) then x-=2 end
if btn(1) then x+=2 end
if btn(2) then y-=2 end
if btn(3) then y+=2 end

-- Spawn multiple enemies
enemies={}
add(enemies,{x=rnd(120),y=0,w=6,h=6})
for e in all(enemies) do e.y+=2 end

-- Score display
score = 0
score +=1
print("score: "..score,2,2,7)

Debugging Tips

  • Use print(var) to inspect values
  • Use cls() to clear screen
  • Draw bounding boxes to check collision
  • Step through one feature at a time
